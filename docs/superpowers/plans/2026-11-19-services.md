# Module 2: Services Catalogue Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Owner and managers keep one list of the services ACE sells (add, edit, reorder, switch off and on). Quotes (Module 3) will be built from it. There are no prices anywhere in it.

**Architecture:** A new `services` table (seeded with the seven starting services by the migration) and plain functions in `packages/core/src/services/` that take a database handle plus an `Actor`, check the role first, and write audit lines holding field names only. The admin app gets one `/services` page whose server actions call those functions inside a transaction. This copies the pattern Module 1 uses for customers.

**Tech Stack:** Next.js 16.4 (App Router, server actions), React 19, TypeScript strict, Prisma 7 with `@prisma/adapter-pg`, Supabase Postgres, zod 4, vitest 5, Tailwind 4.

**Spec:** `docs/superpowers/specs/2026-11-19-services-design.md`

## Global Constraints

- No prices, quote lines, frequencies or agreed scope in this module (those are Module 3). The property fields `agreedMonthlyFeeCents`, `extrasNote` and `frequency` are NOT touched.
- Every new table: Row Level Security enabled, no policies, privileges revoked from `anon` and `authenticated`. Only server code reads or writes.
- Every protected page and server action calls `requireStaff(["OWNER", "MANAGER"])` (the `MANAGEMENT` roles) itself first. Role rules are also enforced in the core functions with `assertRole(actor, MANAGEMENT)` as the first line. CREW get no access in this module.
- Services are switched off (`archivedAt` set), never deleted.
- **Audit entries hold field names, ids and fixed words only (such as the move direction or the kind), never the name or description that was typed.**
- A save that changes nothing writes nothing: no database write, no `updatedAt` change, no audit line.
- A duplicate name (ignoring upper/lower case, including switched-off services) gives the plain message "A service with that name already exists." and never a crash.
- Client components never import values from `@ace/core`; they import from `@ace/core/client` (browser-safe).
- Every page works at phone width (375px) with no sideways scrolling; buttons are at least 40px high or wide.
- Copy rule: no em dashes in any text shown on screen.
- No real client data anywhere. Database tests run only against the DEV database inside always-rolled-back transactions (`npm run test:db`); CI skips them.
- Code style: Prettier (`npm run format`), ESLint clean, `noUncheckedIndexedAccess` is on (index access can be `undefined`).
- Windows: if a terminal is PowerShell, use `npm.cmd run ...` (`npm run` is blocked by execution policy). Git Bash is fine with `npm run`.
- Commit messages end with `Co-Authored-By: Claude Sonnet 5.5 <noreply@anthropic.com>`. Commit locally after each task. Do not push. Do not apply the migration yourself (Task 3 belongs to the controller and needs the owner's approval).
- Next.js 16 specifics already used here: `params` and `searchParams` are Promises (`await` them); server actions live in `"use server"` files that export only async functions.

```ts
import { z } from "zod";

export const SERVICE_KINDS = ["CORE", "OPTIONAL_EXTRA"] as const;
export type ServiceKindValue = (typeof SERVICE_KINDS)[number];

export const SERVICE_KIND_LABELS: Record<ServiceKindValue, string> = {
  CORE: "Core service",
  OPTIONAL_EXTRA: "Optional extra",
};

export const SERVICE_NAME_MAX = 80;
export const SERVICE_DESCRIPTION_MAX = 300;

const tooLong = (max: number) => `Keep this under ${max} characters.`;

/** Form boxes arrive as strings. Both texts are required. */
export const serviceSchema = z.object({
  name: z.string().trim().min(1, "Enter a name.").max(SERVICE_NAME_MAX, tooLong(SERVICE_NAME_MAX)),
  description: z
    .string()
    .trim()
    .min(1, "Enter a description.")
    .max(SERVICE_DESCRIPTION_MAX, tooLong(SERVICE_DESCRIPTION_MAX)),
  kind: z.enum(SERVICE_KINDS, { error: "Choose core service or optional extra." }),
});
export type ServiceInput = z.output<typeof serviceSchema>;
```

- [ ] **Step 1: Write the failing database tests** `packages/core/src/services/services.db.test.ts`

```ts
import { randomUUID } from "node:crypto";
import { describe, expect, it } from "vitest";
import { DomainError, ForbiddenError, type Actor } from "../customers/access";
import { dbTestsEnabled, withRollback } from "../testing/rollback";
import { serviceSchema } from "./schemas";
import {
  archiveService,
  createService,
  getService,
  listServices,
  moveService,
  restoreService,
  updateService,
} from "./services";

const owner: Actor = { id: randomUUID(), role: "OWNER" };
const manager: Actor = { id: randomUUID(), role: "MANAGER" };
const crew: Actor = { id: randomUUID(), role: "CREW" };

const sleep = (ms: number) => new Promise((resolve) => setTimeout(resolve, ms));
// A unique name per call, so nothing collides with the rows already in the dev database.
const token = () => randomUUID().replaceAll("-", "").slice(0, 10);
const input = (over: Record<string, string> = {}) =>
  serviceSchema.parse({
    name: `Test Service ${token()}`,
    description: `Test description ${token()}`,
    kind: "CORE",
    ...over,
  });

describe.skipIf(!dbTestsEnabled)("services data layer", () => {
  it("adds a service at the end and audits it without the typed text", async () => {
    await withRollback(async (tx) => {
      const before = await listServices(tx, manager);
      const data = input();
      const { id } = await createService(tx, manager, data);

      const after = await listServices(tx, manager);
      expect(after).toHaveLength(before.length + 1);
      expect(after.at(-1)?.id).toBe(id);
      expect(after.at(-1)?.kind).toBe("CORE");

      const audit = await tx.auditLog.findMany({ where: { entityId: id } });
      expect(audit.map((a) => a.action)).toEqual(["service.created"]);
      expect(audit[0]?.entityType).toBe("service");
      const text = JSON.stringify(audit);
      expect(text).not.toContain(data.name);
      expect(text).not.toContain(data.description);
    });
  });

  it("refuses a duplicate name ignoring case, including a switched-off service", async () => {
    await withRollback(async (tx) => {
      const data = input({ name: `Dup ${token()}` });
      const { id } = await createService(tx, manager, data);
      await expect(
        createService(tx, manager, { ...data, name: data.name.toUpperCase() }),
      ).rejects.toThrow(DomainError);
      await archiveService(tx, manager, id);
      await expect(
        createService(tx, manager, { ...data, name: data.name.toLowerCase() }),
      ).rejects.toThrow("A service with that name already exists.");
    });
  });

  it("edits a service and audits only the field names", async () => {
    await withRollback(async (tx) => {
      const { id } = await createService(tx, manager, input());
      const edited = input({ kind: "OPTIONAL_EXTRA" });
      await updateService(tx, manager, id, edited);

      const saved = await getService(tx, manager, id);
      expect(saved?.name).toBe(edited.name);
      expect(saved?.kind).toBe("OPTIONAL_EXTRA");

      const audit = await tx.auditLog.findMany({
        where: { entityId: id, action: "service.updated" },
      });
      expect(audit).toHaveLength(1);
      expect(audit[0]?.after).toEqual({ changed: ["name", "description", "kind"] });
      const text = JSON.stringify(audit);
      expect(text).not.toContain(edited.name);
      expect(text).not.toContain(edited.description);
    });
  });

  it("writes nothing when a save changes nothing", async () => {
    await withRollback(async (tx) => {
      const data = input();
      const { id } = await createService(tx, manager, data);
      const before = await getService(tx, manager, id);
      await sleep(20);
      await updateService(tx, manager, id, data);
      const after = await getService(tx, manager, id);
      expect(after?.updatedAt.getTime()).toBe(before?.updatedAt.getTime());
      const updates = await tx.auditLog.count({
        where: { entityId: id, action: "service.updated" },
      });
      expect(updates).toBe(0);
    });
  });

  it("lets a service change only the case of its own name, but not take another's name", async () => {
    await withRollback(async (tx) => {
      const a = input({ name: `Alpha ${token()}` });
      const b = input({ name: `Bravo ${token()}` });
      const { id: aId } = await createService(tx, manager, a);
      await createService(tx, manager, b);
      await updateService(tx, manager, aId, { ...a, name: a.name.toUpperCase() });
      expect((await getService(tx, manager, aId))?.name).toBe(a.name.toUpperCase());
      await expect(
        updateService(tx, manager, aId, { ...a, name: b.name.toLowerCase() }),
      ).rejects.toThrow("A service with that name already exists.");
    });
  });

  it("switches a service off and on; off hides it, on puts it back at the end; repeats do nothing", async () => {
    await withRollback(async (tx) => {
      const { id } = await createService(tx, manager, input());
      await createService(tx, manager, input()); // so "the end" is not already this one

      await archiveService(tx, manager, id);
      expect((await listServices(tx, manager)).some((s) => s.id === id)).toBe(false);
      expect(
        (await listServices(tx, manager, { includeArchived: true })).some((s) => s.id === id),
      ).toBe(true);
      await archiveService(tx, manager, id); // already off: nothing happens
      expect(await tx.auditLog.count({ where: { entityId: id, action: "service.archived" } })).toBe(
        1,
      );

      await restoreService(tx, manager, id);
      const active = await listServices(tx, manager);
      expect(active.at(-1)?.id).toBe(id);
      expect(active.at(-1)?.archivedAt).toBeNull();
      await restoreService(tx, manager, id); // already on: nothing happens
      expect(await tx.auditLog.count({ where: { entityId: id, action: "service.restored" } })).toBe(
        1,
      );
    });
  });

  it("moves a service up and down past its neighbour, and does nothing at the ends", async () => {
    await withRollback(async (tx) => {
      const { id: a } = await createService(tx, manager, input());
      const { id: b } = await createService(tx, manager, input());
      const lastTwo = async () => (await listServices(tx, manager)).slice(-2).map((s) => s.id);
      expect(await lastTwo()).toEqual([a, b]);

      await moveService(tx, manager, b, "up");
      expect(await lastTwo()).toEqual([b, a]);
      await moveService(tx, manager, b, "down");
      expect(await lastTwo()).toEqual([a, b]);

      // The last service cannot go further down, and the first cannot go further up.
      await moveService(tx, manager, b, "down");
      expect(await lastTwo()).toEqual([a, b]);
      const first = (await listServices(tx, manager))[0];
      await moveService(tx, manager, first!.id, "up");
      expect((await listServices(tx, manager))[0]?.id).toBe(first!.id);
      expect(
        await tx.auditLog.count({ where: { entityId: first!.id, action: "service.moved" } }),
      ).toBe(0);

      const moves = await tx.auditLog.findMany({
        where: { entityId: b, action: "service.moved" },
        orderBy: { at: "asc" },
      });
      expect(moves.map((m) => m.after)).toEqual([{ direction: "up" }, { direction: "down" }]);
    });
  });

  it("does not move a switched-off or unknown service", async () => {
    await withRollback(async (tx) => {
      const { id } = await createService(tx, manager, input());
      await archiveService(tx, manager, id);
      await expect(moveService(tx, manager, id, "up")).rejects.toThrow(DomainError);
      await expect(moveService(tx, manager, randomUUID(), "up")).rejects.toThrow(DomainError);
    });
  });

  it("says a missing service could not be found", async () => {
    await withRollback(async (tx) => {
      await expect(updateService(tx, manager, randomUUID(), input())).rejects.toThrow(
        "That service could not be found.",
      );
      await expect(archiveService(tx, manager, randomUUID())).rejects.toThrow(DomainError);
      await expect(restoreService(tx, manager, randomUUID())).rejects.toThrow(DomainError);
      expect(await getService(tx, manager, randomUUID())).toBeNull();
    });
```

```ts
import { assertRole, DomainError, MANAGEMENT, type Actor } from "../customers/access";
import { changedFields } from "../customers/store";
import type { Prisma } from "../generated/prisma/client";
import type { ServiceInput } from "./schemas";
import { logServiceEvent, type ServiceStore } from "./store";

const NOT_FOUND = "That service could not be found.";
const DUPLICATE = "A service with that name already exists.";

/** Display order: by number, ties broken by when the service was added. */
const ORDER: Prisma.ServiceOrderByWithRelationInput[] = [
  { sortOrder: "asc" },
  { createdAt: "asc" },
];

export async function listServices(
  store: ServiceStore,
  actor: Actor,
  opts: { includeArchived?: boolean } = {},
) {
  assertRole(actor, MANAGEMENT);
  return store.service.findMany({
    where: opts.includeArchived ? {} : { archivedAt: null },
    orderBy: ORDER,
  });
}

export async function getService(store: ServiceStore, actor: Actor, id: string) {
  assertRole(actor, MANAGEMENT);
  return store.service.findUnique({ where: { id } });
}

async function findService(store: ServiceStore, id: string) {
  const service = await store.service.findUnique({ where: { id } });
  if (!service) throw new DomainError(NOT_FOUND);
  return service;
}

/**
 * Locks one service row until the transaction ends, by touching it, so two people changing the
 * same service take turns. Call it only when something is really about to be written.
 */
async function lockService(store: ServiceStore, id: string): Promise<void> {
  const { count } = await store.service.updateMany({
    where: { id },
    data: { updatedAt: new Date() },
  });
  if (count !== 1) throw new DomainError(NOT_FOUND);
}

/** Same idea for reordering, which changes several rows at once. */
async function lockSwitchedOnServices(store: ServiceStore): Promise<void> {
  await store.service.updateMany({ where: { archivedAt: null }, data: { updatedAt: new Date() } });
}

/** True when the database refused a write because the name is already taken. */
function isNameClash(error: unknown): boolean {
  if (typeof error !== "object" || error === null) return false;
  const { code, message } = error as { code?: unknown; message?: unknown };
  return (
    code === "P2002" || (typeof message === "string" && message.includes("services_name_lower_key"))
  );
}

async function assertNameFree(store: ServiceStore, name: string, exceptId?: string): Promise<void> {
  const clash = await store.service.findFirst({
    where: {
      name: { equals: name, mode: "insensitive" },
      ...(exceptId ? { id: { not: exceptId } } : {}),
    },
    select: { id: true },
  });
  if (clash) throw new DomainError(DUPLICATE);
}

export async function createService(store: ServiceStore, actor: Actor, input: ServiceInput) {
  assertRole(actor, MANAGEMENT);
  await assertNameFree(store, input.name);
  const last = await store.service.findFirst({
    orderBy: { sortOrder: "desc" },
    select: { sortOrder: true },
  });
  let created: { id: string };
  try {
    created = await store.service.create({
      data: { ...input, sortOrder: (last?.sortOrder ?? 0) + 1 },
      select: { id: true },
    });
  } catch (error) {
    // Two people adding the same name at the same moment: the database index catches the loser.
    if (isNameClash(error)) throw new DomainError(DUPLICATE);
    throw error;
  }
  await logServiceEvent(store, actor, "service.created", created.id, { kind: input.kind });
  return created;
}

export async function updateService(
  store: ServiceStore,
  actor: Actor,
  id: string,
  input: ServiceInput,
): Promise<void> {
  assertRole(actor, MANAGEMENT);
  // Look first WITHOUT locking: a save that changes nothing must write nothing.
  if (changedFields(await findService(store, id), input).length === 0) return;
  // Something changed: lock, then read again, because a concurrent edit may have landed between.
  await lockService(store, id);
  const existing = await findService(store, id);
  const changed = changedFields(existing, input);
  if (changed.length === 0) return;
  if (changed.includes("name")) await assertNameFree(store, input.name, id);
  try {
    await store.service.update({ where: { id }, data: input });
  } catch (error) {
    if (isNameClash(error)) throw new DomainError(DUPLICATE);
    throw error;
  }
  await logServiceEvent(store, actor, "service.updated", id, { changed });
}

export async function archiveService(store: ServiceStore, actor: Actor, id: string): Promise<void> {
  assertRole(actor, MANAGEMENT);
  if ((await findService(store, id)).archivedAt) return;
  await lockService(store, id);
  if ((await findService(store, id)).archivedAt) return;
  await store.service.update({ where: { id }, data: { archivedAt: new Date() } });
  await logServiceEvent(store, actor, "service.archived", id);
}

/** Switching a service back on puts it at the end of the list. */
export async function restoreService(store: ServiceStore, actor: Actor, id: string): Promise<void> {
  assertRole(actor, MANAGEMENT);
  if (!(await findService(store, id)).archivedAt) return;
  await lockService(store, id);
  if (!(await findService(store, id)).archivedAt) return;
  const last = await store.service.findFirst({
    orderBy: { sortOrder: "desc" },
    select: { sortOrder: true },
  });
  await store.service.update({
    where: { id },
    data: { archivedAt: null, sortOrder: (last?.sortOrder ?? 0) + 1 },
  });
  await logServiceEvent(store, actor, "service.restored", id);
}

/**
 * Swaps a switched-on service with its neighbour (up or down) and renumbers the switched-on
 * services 1, 2, 3... At the top or bottom there is nothing to do and nothing is written.
 */
export async function moveService(
  store: ServiceStore,
  actor: Actor,
  id: string,
  direction: "up" | "down",
): Promise<void> {
  assertRole(actor, MANAGEMENT);
  const switchedOn = () =>
    store.service.findMany({
      where: { archivedAt: null },
      orderBy: ORDER,
      select: { id: true, sortOrder: true },
    });
  /** The new order of ids, or null when the service is already at that end. */
  const plan = (list: { id: string }[]): string[] | null => {
    const from = list.findIndex((s) => s.id === id);
    if (from === -1) throw new DomainError("Only a switched-on service can be moved.");
    const to = direction === "up" ? from - 1 : from + 1;
    if (to < 0 || to >= list.length) return null;
    const ids = list.map((s) => s.id);
    const moved = ids[from];
    const other = ids[to];
    if (moved === undefined || other === undefined) return null;
    ids[from] = other;
    ids[to] = moved;
    return ids;
  };

  // Look first WITHOUT locking, so a move that cannot happen leaves everything untouched.
  if (plan(await switchedOn()) === null) return;
  await lockSwitchedOnServices(store);
  const list = await switchedOn();
  const order = plan(list);
  if (order === null) return;
  const current = new Map(list.map((s) => [s.id, s.sortOrder]));
  for (const [index, serviceId] of order.entries()) {
    if (current.get(serviceId) === index + 1) continue;
    await store.service.update({ where: { id: serviceId }, data: { sortOrder: index + 1 } });
  }
  await logServiceEvent(store, actor, "service.moved", id, { direction });
}
```

```ts
"use server";

import {
  archiveService,
  createService,
  inTx,
  MANAGEMENT,
  moveService,
  restoreService,
  serviceSchema,
  updateService,
} from "@ace/core";
import { redirect } from "next/navigation";
import { z } from "zod";
import { toActor } from "@/lib/actor";
import { requireStaff } from "@/lib/auth";
import { errorsFrom, failure, formValues, type FormState } from "@/lib/form";

const idSchema = z.uuid();
const direction = z.enum(["up", "down"]);

/** Add (no serviceId) or edit (with serviceId). */
export async function saveServiceAction(_prev: FormState, formData: FormData): Promise<FormState> {
  const staff = await requireStaff(MANAGEMENT);
  const actor = await toActor(staff);
  const values = formValues(formData);
  const serviceId = values.serviceId ? idSchema.safeParse(values.serviceId) : null;
  if (serviceId && !serviceId.success) {
    return { error: "Something went wrong. Refresh the page and try again.", nonce: Date.now() };
  }

  const parsed = serviceSchema.safeParse(values);
  if (!parsed.success) return { fieldErrors: errorsFrom(parsed.error), values, nonce: Date.now() };

  try {
    await inTx(async (tx) => {
      if (serviceId?.success) await updateService(tx, actor, serviceId.data, parsed.data);
      else await createService(tx, actor, parsed.data);
    });
  } catch (error) {
    return failure(error, values);
  }
  redirect("/services");
}

export async function archiveServiceAction(formData: FormData): Promise<void> {
  const staff = await requireStaff(MANAGEMENT);
  const actor = await toActor(staff);
  const id = idSchema.parse(formData.get("id"));
  try {
    await inTx((tx) => archiveService(tx, actor, id));
  } catch (error) {
    failure(error);
  }
  redirect("/services");
}

export async function restoreServiceAction(formData: FormData): Promise<void> {
  const staff = await requireStaff(MANAGEMENT);
  const actor = await toActor(staff);
  const id = idSchema.parse(formData.get("id"));
  try {
    await inTx((tx) => restoreService(tx, actor, id));
  } catch (error) {
    failure(error);
  }
  redirect("/services?archived=1");
}

  const actor = await toActor(staff);
  const id = idSchema.parse(formData.get("id"));
  const dir = direction.parse(formData.get("direction"));
  try {
    await inTx((tx) => moveService(tx, actor, id, dir));
  } catch (error) {
    failure(error);
  }
  redirect("/services");
}
```
